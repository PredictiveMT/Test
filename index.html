fetch("data.json") .then(res=>res.json()) .then(data=>{ const chart = echarts.init( document.getElementById('healthChart') ); chart.setOption({ tooltip:{}, xAxis:{ type:'category', data:data.devices.map( x=>x.device ) }, yAxis:{ max:100 }, series:[{ type:'bar', data:data.devices.map( x=>x.health ), itemStyle:{ color:function(params){ let v=params.value; if(v<70) return "#ff4d4f"; if(v<85) return "#faad14"; return "#52c41a"; } } }] }); });
